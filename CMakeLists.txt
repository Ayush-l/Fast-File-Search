cmake_minimum_required(VERSION 3.16)
project(FastFileSearcherCPP VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This desktop application targets Windows (Win32 GUI).")
endif()
add_executable(FastFileSearcher WIN32 src/main.cpp src/SearchEngine.cpp src/SearchEngine.h)
target_compile_features(FastFileSearcher PRIVATE cxx_std_17)
target_compile_definitions(FastFileSearcher PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(FastFileSearcher PRIVATE comctl32 shell32 ole32)
if(MSVC)
  target_compile_options(FastFileSearcher PRIVATE /W4 /EHsc)
else()
  target_compile_options(FastFileSearcher PRIVATE -Wall -Wextra)
endif()
